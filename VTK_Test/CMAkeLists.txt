cmake_minimum_required(VERSION 3.20)

project(
    DataVisualizer
    VERSION 1.0.0
    DESCRIPTION "Geophysical Data Visualization Tool"
    LANGUAGES CXX
)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)

option(DATAVIZ_STRICT_WARNINGS "Enable stricter compiler warnings" ON)
set(DATAVIZ_QT_ROOT "" CACHE PATH "Optional path to Qt6 root (for example C:/Qt/6.11.0/mingw_64)")

if(DATAVIZ_QT_ROOT)
    list(PREPEND CMAKE_PREFIX_PATH "${DATAVIZ_QT_ROOT}")
endif()

find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets PrintSupport)


find_package(VTK REQUIRED COMPONENTS
    CommonCore
    CommonDataModel
    FiltersCore
    FiltersGeometry      
    RenderingCore
    RenderingOpenGL2
    InteractionStyle
    RenderingFreeType
    GUISupportQt      
)

set(SOURCES
    src/main.cpp
    src/MainWindow.cpp
    src/PlotWidget.cpp
    src/DataParser.cpp
    src/core/AppConfig.cpp
    src/core/Core.cpp
    src/core/GridModel.cpp
    src/coords/CoordTransformer.cpp
    src/coords/FortrenBridge.cpp
    src/coords/MagneticCoords.cpp
    src/ui/ControlPanel.cpp
    src/visualization/PlotEngine.cpp
    src/visualization/PolarPlot.cpp
    src/visualization/Colormaps.cpp
    src/visualization/Overlays.cpp
    src/visualization/ContourPlot.cpp
    src/visualization/VectorPlot.cpp
)

set(HEADERS
    src/MainWindow.h
    src/PlotWidget.h
    src/DataParser.h
    src/core/AppConfig.h
    src/core/Core.h
    src/core/GridModel.h
    src/coords/CoordTransformer.h
    src/coords/FortranBridge.h
    src/coords/MagneticCoords.h
    src/ui/ControlPanel.h
    src/visualization/PlotEngine.h
    src/visualization/PolarPlot.h
    src/visualization/Colormaps.h
    src/visualization/Overlays.h
    src/visualization/ContourPlot.h
    src/visualization/VectorPlot.h
)

add_executable(${PROJECT_NAME} ${SOURCES} ${HEADERS})

target_include_directories(${PROJECT_NAME} PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/external/eigen
    ${CMAKE_CURRENT_SOURCE_DIR}/src
)


target_link_libraries(${PROJECT_NAME}
    PRIVATE
        Qt6::Core
        Qt6::Gui
        Qt6::Widgets
        Qt6::PrintSupport
        VTK::CommonCore
        VTK::CommonDataModel
        VTK::FiltersCore
        VTK::FiltersGeometry
        VTK::RenderingCore
        VTK::RenderingOpenGL2
        VTK::InteractionStyle
        VTK::RenderingFreeType
        VTK::GUISupportQt  
)


vtk_module_autoinit(
    TARGETS ${PROJECT_NAME}
    MODULES VTK::RenderingOpenGL2 VTK::InteractionStyle
)

if(DATAVIZ_STRICT_WARNINGS)
    if(MSVC)
        target_compile_options(${PROJECT_NAME} PRIVATE /W4 /utf-8 /EHsc)
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()

if(WIN32)
    set_target_properties(${PROJECT_NAME} PROPERTIES
        WIN32_EXECUTABLE TRUE
        RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin
    )
endif()

install(TARGETS ${PROJECT_NAME}
    RUNTIME DESTINATION bin
    LIBRARY DESTINATION lib
    ARCHIVE DESTINATION lib
)
